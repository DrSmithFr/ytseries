> // series.model.ts
> import { SeasonDto, SeasonMapper, SeasonModel } from './season.model';
> import { AbstractMapper } from '../abstract-mapper';
> import { inject } from '@angular/core';
>
> export interface SeriesDto {
>   id?: number;
>   import_code?: string;
>   name?: string;
>   image?: string;
>   locale?: string;
>   tags?: string[];
>   description?: string;
>   seasons?: SeasonDto[];
>   created_at?: string;
> }
>
> export class SeriesModel {
>   constructor(
>     public id: number,
>     public import_code: string,
>     public name: string,
>     public image: string,
>     public locale: string,
>     public tags: string[],
>     public description: string,
>     public seasons: SeasonModel[],
>     public created_at: Date
>   ) {}
> }
>
> export class SeriesMapper extends AbstractMapper<SeriesDto, SeriesModel> {
>   override fromDto(dto: SeriesDto): SeriesModel {
>     return new SeriesModel(
>       this.mapNumber(dto.id),
>       this.mapString(dto.import_code),
>       this.mapString(dto.name),
>       this.mapString(dto.image),
>       this.mapString(dto.locale),
>       this.mapArray<string>(dto.tags, this.mapString),
>       this.mapString(dto.description),
>       this.mapModels(dto.seasons, inject(SeasonMapper)),
>       this.mapDate(dto.created_at)
>     );
>   }
>
>   override toDto(model: SeriesModel): SeriesDto {
>     return {
>       id: model.id,
>       import_code: model.import_code,
>       name: model.name,
>       image: model.image,
>       locale: model.locale,
>       tags: model.tags,
>       description: model.description,
>       seasons: inject(SeasonMapper).toDtos(model.seasons),
>       created_at: model.created_at.toISOString()
>     } as SeriesDto;
>   }
> }
> ```

**Notes (Angular/TypeScript context):**
- **Embedded entities** (e.g., `SeasonModel` inside `SeriesModel`):
  - The prompt will auto‑generate the corresponding file (`season.model.ts`).
  - The mapper leverages Angular’s `inject()` to obtain the child mapper, ensuring proper dependency injection.
- **Arrays**:
  - In the mapper use `mapArray<T>(array: T[], mapper: (item: T) => any)` to transform list items.
- **Dates**:
  - `mapDate(dto.created_at)` turns a string (or timestamp) into a `Date` object.
  - `model.created_at.toISOString()` serialises the `Date` back to an ISO‑8601 string for the DTO.
***Mapping function (used in fromDto())
- `mapString`, `mapNumber`, `mapBoolean` for simple types.
- `mapArray<T>(values, fn)` for transforming lists with custom mappers.
- `mapModel(value, mapper)` and `mapModels(values, mapper)` for nested entities.
- `mapDate(value)` for ISO string → `Date` conversion.
- Emphasize that these helpers should be used in `fromDto()` and `toDto()` as shown in examples.
